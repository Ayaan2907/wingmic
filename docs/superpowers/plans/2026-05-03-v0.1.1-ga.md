      .update(schema.companies)
      .set({

        industry: mergedIndustry,
        domain: existing.domain ?? c.domainHint ?? null,

        promotedAt:
          nextCount >= 2 && !existing.promotedAt ? new Date() : existing.promotedAt,
      })
      .where(eq(schema.companies.id, existing.id));
    return existing.id;
  }

    .insert(schema.companies)
    .values({
      slug,
      name: c.name,
      domain: c.domainHint ?? null,
      industry: c.industry,
      observedCount: 1,
    })
    .returning({ id: schema.companies.id });

function unionDedup(a: string[], b: string[]): string[] {
  return Array.from(new Set([...a, ...b])).sort();

  db: DB,
  e: EventCandidate,
  capturedAt: Date,
): Promise<string> {
  const slug = slugify(e.name);
  const dateGuess =
    e.dateHint && /^\d{4}-\d{2}-\d{2}/.test(e.dateHint)
      ? new Date(e.dateHint)
      : capturedAt;

  const existing = await db.query.events.findFirst({
    where: eq(schema.events.slug, slug),
  });

  if (existing) {
    const nextCount = existing.observedCount + 1;
    const candidates = [...(existing.locationCandidates ?? [])];
    if (e.location) candidates.push(e.location);
    const winningLocation = mode(candidates) ?? existing.location;
    await db
      .update(schema.events)
      .set({
        observedCount: nextCount,
        locationCandidates: candidates,
        location: winningLocation,
        promotedAt:
          nextCount >= 2 && !existing.promotedAt ? new Date() : existing.promotedAt,
      })
      .where(eq(schema.events.id, existing.id));
    return existing.id;
  }

  const inserted = await db

    .values({

      dateRangeStart: dateGuess,
      dateRangeEnd: dateGuess,
      location: e.location ?? null,
      locationCandidates: e.location ? [e.location] : [],
      observedCount: 1,
    })
    .returning({ id: schema.events.id });
  return inserted[0].id;

function mode<T extends string>(values: T[]): T | null {
  if (values.length === 0) return null;
  const counts = new Map<T, number>();
  let best: T = values[0];
  let bestCount = 0;
  for (const v of values) {
    const next = (counts.get(v) ?? 0) + 1;
    counts.set(v, next);
    if (next > bestCount) {
      best = v;
      bestCount = next;
    }
  }
  return best;
}

- [ ] **Step 3: Add the PostHog wrapper**

Create `apps/web/lib/observability/posthog.ts`:

```typescript
import { PostHog } from 'posthog-node';

let client: PostHog | null = null;
let initialized = false;

function getClient(): PostHog | null {
  if (initialized) return client;
  initialized = true;
  const key = process.env.POSTHOG_KEY;
  const host = process.env.POSTHOG_HOST ?? 'https://us.i.posthog.com';
  if (!key) return null;
  client = new PostHog(key, { host, flushAt: 1, flushInterval: 0 });
  return client;
}

export type CaptureEvent =
  | 'capture_started'
  | 'capture_committed'
  | 'recall_query'
  | 'extraction_failed';

interface EventProps {
  durationMs?: number;
  newEntities?: number;
  matchedEntities?: number;
  pendingMatches?: number;
  resultCount?: number;
  errorClass?: string;
  modelVersion?: string;
}

export async function capture(
  event: CaptureEvent,
  distinctId: string,
  props: EventProps = {},
): Promise<void> {
  const c = getClient();
  if (!c) return;
  c.capture({
    distinctId,
    event,
    properties: {
      ...props,
      $process_person_profile: false, // anonymous-ish, no profile
    },
  });
  await c.shutdown(); // ensure flush in serverless context
}
```
